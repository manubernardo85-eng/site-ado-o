    <div>
      <button class="yes" onclick="votar('sim')">💖 Adotar</button>
      <button class="no" onclick="votar('nao')">💔 Não adotar</button>
    </div>

    <canvas id="grafico"></canvas>
  </div>

<script>
  let votosSim = 0;
  let votosNao = 0;

  const ctx = document.getElementById("grafico");

  const chart = new Chart(ctx, {
    type: "pie",
    data: {
      labels: ["Adotar", "Não adotar"],
      datasets: [{
        data: [votosSim, votosNao]
      }]
    },
    options: {
      plugins: {
        legend: { position: "bottom" }
      }
    }
  });

  function votar(tipo) {
    if (tipo === "sim") votosSim++;
    if (tipo === "nao") votosNao++;

    chart.data.datasets[0].data = [votosSim, votosNao];
    chart.update();
  }
</script>
